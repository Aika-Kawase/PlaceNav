cmake_minimum_required(VERSION 3.0.2)
project(source_costmaps)
add_compile_options(-std=c++14 -Wall -Wextra -Wpedantic)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs geometry_msgs nav_msgs diagnostic_msgs tf2 tf2_ros costmap_2d)
catkin_package(INCLUDE_DIRS include LIBRARIES source_costmaps_cloud CATKIN_DEPENDS roscpp sensor_msgs std_msgs geometry_msgs nav_msgs diagnostic_msgs tf2 tf2_ros costmap_2d)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(source_costmaps_cloud src/cloud.cpp)
target_link_libraries(source_costmaps_cloud ${catkin_LIBRARIES})
add_executable(cloud_gate_node src/cloud_gate_node.cpp)
target_link_libraries(cloud_gate_node source_costmaps_cloud ${catkin_LIBRARIES})
add_executable(source_costmaps_node src/source_costmaps_node.cpp)
target_link_libraries(source_costmaps_node ${catkin_LIBRARIES})
catkin_install_python(PROGRAMS scripts/decompress_c2.py scripts/cloud_batch.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS source_costmaps_cloud cloud_gate_node source_costmaps_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_add_gtest(test_cloud test/test_cloud.cpp)
  if(TARGET test_cloud)
    target_link_libraries(test_cloud source_costmaps_cloud ${catkin_LIBRARIES})
  endif()
  catkin_install_python(PROGRAMS test/test_layers.py test/test_comparison.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  add_rostest(test/layers.test)
  add_rostest(test/comparison.test)
  add_rostest(test/comparison_frames.test)
endif()
